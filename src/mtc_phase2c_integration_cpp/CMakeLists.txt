cmake_minimum_required(VERSION 3.16)
project(mtc_phase2c_integration_cpp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(mtc_trajectory_execution_cpp REQUIRED)
find_package(mtc_moveit_planner REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(moveit_core REQUIRED)
add_library(mtc_planning_only_manager SHARED src/planning_only_manager.cpp)
target_include_directories(mtc_planning_only_manager PUBLIC include)
ament_target_dependencies(mtc_planning_only_manager pluginlib moveit_core)
pluginlib_export_plugin_description_file(moveit_core planning_only_plugins.xml)
install(TARGETS mtc_planning_only_manager LIBRARY DESTINATION lib)
add_library(${PROJECT_NAME} SHARED src/dispatch_adapter.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} PUBLIC mtc_trajectory_execution_cpp)
add_executable(single_owner src/single_owner.cpp)
target_link_libraries(single_owner ${PROJECT_NAME})
ament_target_dependencies(single_owner mtc_moveit_planner rclcpp rclcpp_action control_msgs)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(TARGETS single_owner DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/run_fake_gate.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_dispatch_adapter test/test_dispatch_adapter.cpp)
  target_link_libraries(test_dispatch_adapter ${PROJECT_NAME})
  ament_add_gtest(test_planning_only_manager test/test_planning_only_manager.cpp)
  target_include_directories(test_planning_only_manager PRIVATE include)
  ament_target_dependencies(test_planning_only_manager moveit_core pluginlib)
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(mtc_trajectory_execution_cpp moveit_core pluginlib)
ament_package()
