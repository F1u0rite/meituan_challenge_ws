cmake_minimum_required(VERSION 3.16)
project(mtc_aubo_fake_integration_cpp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep mtc_trajectory_execution_cpp mtc_moveit_planner rclcpp rclcpp_action sensor_msgs control_msgs moveit_core moveit_ros_planning pluginlib nlohmann_json urdf)
  find_package(${dep} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/aubo_fjt_motion_port.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} mtc_trajectory_execution_cpp::mtc_trajectory_execution_cpp mtc_moveit_planner::mtc_moveit_planner_core)
ament_target_dependencies(${PROJECT_NAME} rclcpp)
add_library(mtc_aubo_planning_only SHARED src/planning_only.cpp)
ament_target_dependencies(mtc_aubo_planning_only pluginlib moveit_core)
pluginlib_export_plugin_description_file(moveit_core planning_only.xml)
add_executable(aubo_fake_probe src/aubo_fake_probe.cpp)
target_link_libraries(aubo_fake_probe ${PROJECT_NAME} mtc_moveit_planner::mtc_moveit_planner_moveit nlohmann_json::nlohmann_json)
ament_target_dependencies(aubo_fake_probe rclcpp rclcpp_action sensor_msgs control_msgs moveit_core moveit_ros_planning urdf)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib INCLUDES DESTINATION include)
install(TARGETS mtc_aubo_planning_only LIBRARY DESTINATION lib)
install(TARGETS aubo_fake_probe DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/run_aubo_fake.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_aubo_command test/test_aubo_command.cpp TIMEOUT 30)
  target_link_libraries(test_aubo_command ${PROJECT_NAME})
  ament_add_gtest(test_aubo_action test/test_aubo_action.cpp TIMEOUT 90)
  target_include_directories(test_aubo_action PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../mtc_trajectory_execution_cpp/test)
  target_link_libraries(test_aubo_action ${PROJECT_NAME})
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(mtc_trajectory_execution_cpp mtc_moveit_planner rclcpp)
ament_package()
